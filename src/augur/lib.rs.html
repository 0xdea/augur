23/// IDA string type that holds strings extracted from IDA's string list.
24#[derive(Debug, Clone, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
25struct IDAString(String);
26
27impl IDAString {
28    /// Iteratively traverses XREFs and dumps related pseudocode and type definitions to the output file.
29    ///
30    /// # Errors
31    ///
32    /// Returns the appropriate [`HaruspexError`] if the output file cannot be created, the function cannot be
33    /// decompiled, or if the Hex-Rays decompiler license is not available for the target binary.
34    #[expect(
35        clippy::arithmetic_side_effects,
36        reason = "`usize` can hardly overflow here"
37    )]
38    fn traverse_xrefs(
39        &self,
40        idb: &IDB,
41        first_xref: XRef<'_>,
42        addr: Address,
43        dirpath: &Path,
44        string_uses_count: &mut usize,
45    ) -> Result<(), HaruspexError> {
46        let string_name = self.filter_printable_chars();
47        let dirpath_sub = dirpath.join(format!("_{addr:X}_{}_", sanitize_filename(&string_name)));
48
49        let mut current = Some(first_xref);
50
51        while let Some(xref) = current {
52            let from = xref.from();
53
54            // If XREF is in a function, dump the function's pseudocode and type definitions,
55            // otherwise only print its address.
56            if let Some(f) = idb.function_at(from) {
57                // Skip the function if it has the `thunk` attribute.
58                if !f.flags().contains(FunctionFlags::THUNK) {
59                    dump_function_pseudocode(idb, &f, from, &dirpath_sub)?;
60                    *string_uses_count += 1;
61                }
62            } else {
63                println!("{from:#X} in [unknown]");
64            }
65            current = xref.next_to();
66        }
67
68        Ok(())
69    }
70
71    /// Takes an [`IDAString`] as input and returns a [`String`] that contains only its printable chars.
72    fn filter_printable_chars(&self) -> String {
73        self.chars()
74            .filter(|c| c.is_ascii_graphic() || *c == ' ')
75            .collect()
76    }
77}

98/// Extracts strings and pseudocode/type definitions of each function that references them from the
99/// binary at `filepath` and saves them in `filepath.str`.
100///
101/// Returns the number of locations where strings are referenced.
102///
103/// # Errors
104///
105/// Returns [`anyhow::Error`] in case something goes wrong with analyzing the binary file or decompiling functions.
106pub fn run(filepath: impl AsRef<Path>) -> anyhow::Result<usize> {
107    let start = Instant::now();
108
109    eprintln!(
110        "[*] Analyzing binary file `{}`",
111        filepath.as_ref().display()
112    );
113    let mut idb = IDB::open(&filepath).with_context(|| {
114        format!(
115            "Failed to analyze binary file `{}`",
116            filepath.as_ref().display()
117        )
118    })?;
119    eprintln!("[+] Successfully analyzed binary file");
120    eprintln!();
121
122    eprintln!("[-] Processor: {}", idb.processor().long_name());
123    eprintln!("[-] Compiler: {:?}", idb.meta().cc_id());
124    eprintln!("[-] File type: {:?}", idb.meta().filetype());
125    eprintln!();
126
127    anyhow::ensure!(idb.decompiler_available(), "Decompiler is not available");
128
129    // Disable argument name hints.
130    idb.modify_decompiler_config(ArgHintsMode::Disabled.directive())
131        .context("Failed to set decompiler's argument hints mode")?;
132
133    // Create a new output directory, returning an error if it already exists and it's not empty.
134    let dirpath = filepath.as_ref().with_extension("str");
135    prepare_output_dir(&dirpath)?;
136
137    // Leverage the full power of IDA to recover strings during decompilation.
138    eprintln!();
139    eprintln!("[*] Decompiling all functions and recovering strings...");
140    // Cleanup and return an error if the Hex-Rays decompiler license is not available.
141    if let Err(e) = recover_strings(&mut idb) {
142        fs::remove_dir_all(&dirpath)?;
143        return Err(e.into());
144    }
145
146    let mut string_uses_count = 0;
147
148    eprintln!();
149    eprintln!("[*] Finding cross-references to strings...");
150    let strings = idb.strings();
151    for i in 0..strings.len() {
152        // Extract string with its address
153        let string = IDAString::from(
154            strings
155                .get_by_index(i)
156                .context("Failed to get string content")?,
157        );
158        let addr = strings
159            .get_address_by_index(i)
160            .context("Failed to get string address")?;
161        println!("\n{addr:#X} {:?} ", string.as_ref());
162
163        // Traverse XREFs to string and dump the related pseudocode and type definitions to the output files.
164        idb.first_xref_to(addr, XRefQuery::ALL)
165            .map_or(Ok::<(), HaruspexError>(()), |xref| {
166                match string.traverse_xrefs(&idb, xref, addr, &dirpath, &mut string_uses_count) {
167                    // Cleanup and return an error if Hex-Rays decompiler license is not available.
168                    Err(HaruspexError::DecompileFailed(IDAError::HexRays(e)))
169                        if e.code() == HexRaysErrorCode::License =>
170                    {
171                        fs::remove_dir_all(&dirpath)?;
172                        Err(IDAError::HexRays(e).into())
173                    }
174
175                    // Ignore other IDA errors and do nothing when XREF processing is finished.
176                    Err(HaruspexError::DecompileFailed(_)) | Ok(()) => Ok(()),
177
178                    // Propagate any other error.
179                    Err(e) => Err(e),
180                }
181            })?;
182    }
183
184    if string_uses_count == 0 {
185        fs::remove_dir_all(&dirpath)
186            .with_context(|| format!("Failed to remove directory `{}`", dirpath.display()))?;
187        anyhow::bail!("No string uses were found, check your input file");
188    }
189
190    eprintln!();
191    eprintln!(
192        "[+] Found {string_uses_count} string uses in functions, decompiled into `{}`",
193        dirpath.display()
194    );
195    eprintln!(
196        "[+] Done processing binary file `{}` in {:.1} seconds",
197        filepath.as_ref().display(),
198        start.elapsed().as_secs_f64()
199    );
200    Ok(string_uses_count)
201}

203/// Decompiles all functions in the IDB to let IDA 9.4 recover additional strings, ignoring decompilation
204/// errors, then rebuilds the string list.
205///
206/// # Errors
207///
208/// Returns an [`IDAError`] if the Hex-Rays decompiler license is not available for the target binary.
209fn recover_strings(idb: &mut IDB) -> Result<(), IDAError> {
210    for (_id, f) in idb.functions() {
211        if f.flags().contains(FunctionFlags::THUNK) {
212            continue;
213        }
214
215        // Bail out early if the Hex-Rays decompiler license is not available, ignore other IDA errors.
216        if let Err(IDAError::HexRays(e)) = idb.decompile(&f)
217            && e.code() == HexRaysErrorCode::License
218        {
219            return Err(IDAError::HexRays(e));
220        }
221    }
222
223    // The decompiler queues new string items for auto-analysis, so wait for it before rebuilding.
224    if !idb.auto_wait() {
225        eprintln!("[!] Auto-analysis failed");
226    }
227    idb.strings().rebuild();
228
229    Ok(())
230}

232/// Dumps pseudocode of `func` into `dirpath` and prints XREF address, function name, and output path on success.
233///
234/// Alongside the `.c` pseudocode file, a sibling `.h` file with `func`'s type definitions is written
235/// when any are available; if there are none, only the `.c` file is produced.
236///
237/// # Errors
238///
239/// Returns [`HaruspexError`] if the output file cannot be created or the function cannot be decompiled.
240fn dump_function_pseudocode(
241    idb: &IDB,
242    func: &Function<'_>,
243    from: Address,
244    dirpath: &Path,
245) -> Result<(), HaruspexError> {
246    let func_name = func.name().unwrap_or_else(|| "[no name]".into());
247    let output_path = output_path_for_function(func, dirpath);
248
249    fs::create_dir_all(dirpath)?;
250
251    match decompile_to_file(idb, func, &output_path) {
252        // Pseudocode and type definitions were successfully written to the output files.
253        Ok(()) => {
254            println!(
255                "{from:#X} in {func_name} -> `{}` + `{}`",
256                output_path.display(),
257                output_path
258                    .with_extension("h")
259                    .file_name()
260                    .map(OsStr::to_string_lossy)
261                    .unwrap_or_default()
262            );
263            Ok(())
264        }
265
266        // Pseudocode was written, but there were no type definitions to dump.
267        Err(HaruspexError::TypesEmpty) => {
268            println!("{from:#X} in {func_name} -> `{}`", output_path.display());
269            Ok(())
270        }
271
272        // Propagate any other error.
273        Err(e) => Err(e),
274    }
275}

278mod tests {

281    #[test]
282    fn filter_printable_chars_keeps_ascii_graphic_chars() {
283        let s = IDAString::from("hello!@#$%^&*()".to_owned());
284        assert_eq!(
285            s.filter_printable_chars(),
286            "hello!@#$%^&*()",
287            "ascii graphic chars should be kept"
288        );
289    }
290
291    #[test]
292    fn filter_printable_chars_keeps_space() {
293        let s = IDAString::from("hello world".to_owned());
294        assert_eq!(
295            s.filter_printable_chars(),
296            "hello world",
297            "space should be kept"
298        );
299    }
300
301    #[test]
302    fn filter_printable_chars_strips_control_chars() {
303        let s = IDAString::from("hel\x00lo\x01\x1f".to_owned());
304        assert_eq!(
305            s.filter_printable_chars(),
306            "hello",
307            "control chars should be stripped"
308        );
309    }
310
311    #[test]
312    fn filter_printable_chars_strips_nul_bytes() {
313        let s = IDAString::from("foo\x00bar".to_owned());
314        assert_eq!(
315            s.filter_printable_chars(),
316            "foobar",
317            "nul bytes should be stripped"
318        );
319    }
320
321    #[test]
322    fn filter_printable_chars_strips_non_ascii() {
323        let s = IDAString::from("caf\u{00e9}".to_owned());
324        assert_eq!(
325            s.filter_printable_chars(),
326            "caf",
327            "non-ascii chars should be stripped"
328        );
329    }
330
331    #[test]
332    fn filter_printable_chars_on_empty_string_produces_empty_string() {
333        let s = IDAString::from(String::new());
334        assert_eq!(
335            s.filter_printable_chars(),
336            "",
337            "empty input should produce empty output"
338        );
339    }
340
341    #[test]
342    fn filter_printable_chars_on_all_non_printable_chars_produces_empty_string() {
343        let s = IDAString::from("\x00\x01\x02\x03".to_owned());
344        assert_eq!(
345            s.filter_printable_chars(),

347            "all non-printable chars should produce empty string"
